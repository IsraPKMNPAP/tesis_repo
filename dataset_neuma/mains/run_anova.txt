eeg_cols = [c for c in df_wide.columns if re.match(r"^ch\d+_(theta|alpha|beta|gamma)_(mean|std|rel)$", c)]

df = df_wide[[label_col] + eeg_cols].copy()

# Mantener solo filas con etiqueta definida y features no nulas
df = df.dropna(subset=[label_col])
# Opcional: imputación simple si hay NaNs en features (recomendable para EDA)
df[eeg_cols] = df[eeg_cols].fillna(df[eeg_cols].median(numeric_only=True))

from scipy.stats import f_oneway
from statsmodels.stats.multitest import multipletests

def anova_univariate(df, label_col, feature_cols):
    y = df[label_col].values
    g0 = df.loc[y == 0]
    g1 = df.loc[y == 1]

    rows = []
    for feat in feature_cols:
        x0 = g0[feat].values
        x1 = g1[feat].values

        # ANOVA (2 grupos -> equivalente a t-test en hipótesis, pero reportamos como ANOVA)
        F, p = f_oneway(x0, x1)

        # Tamaño de efecto eta^2 = SSB / SST
        overall_mean = df[feat].mean()
        ss_between = len(x0) * (x0.mean() - overall_mean)**2 + len(x1) * (x1.mean() - overall_mean)**2
        ss_total = ((df[feat] - overall_mean)**2).sum()
        eta2 = float(ss_between / ss_total) if ss_total > 0 else 0.0

        rows.append((feat, F, p, eta2, x0.mean(), x1.mean()))

    out = pd.DataFrame(rows, columns=["feature", "F", "p", "eta2", "mean_b0", "mean_b1"])

    # Corrección por múltiples comparaciones (FDR-BH)
    out["p_fdr"] = multipletests(out["p"].values, method="fdr_bh")[1]
    return out.sort_values("eta2", ascending=False)

anova_df = anova_univariate(df, label_col, eeg_cols)
anova_df.head(20)

pattern = re.compile(r"^(ch\d+)_(theta|alpha|beta|gamma)_(mean|std|rel)$")

def split_feature_name(name):
    m = pattern.match(name)
    if not m:
        return (None, None, None)
    return m.group(1), m.group(2), m.group(3)

anova_df[["channel", "band", "metric"]] = anova_df["feature"].apply(
    lambda s: pd.Series(split_feature_name(s))
)

# Resumen por canal: (i) max eta2, (ii) promedio eta2, (iii) cuántas features pasan FDR
chan_summary = (
    anova_df
    .groupby("channel")
    .agg(
        eta2_max=("eta2", "max"),
        eta2_mean=("eta2", "mean"),
        n_sig_fdr=("p_fdr", lambda x: int((x < 0.05).sum())),
        n_features=("feature", "count"),
    )
    .sort_values(["eta2_max", "eta2_mean"], ascending=False)
)

# Resumen por canal+banda (para ver si theta domina)
chan_band_summary = (
    anova_df
    .groupby(["channel", "band"])
    .agg(
        eta2_max=("eta2", "max"),
        eta2_mean=("eta2", "mean"),
        n_sig_fdr=("p_fdr", lambda x: int((x < 0.05).sum())),
        n_features=("feature", "count"),
    )
    .sort_values(["eta2_max", "eta2_mean"], ascending=False)
)
